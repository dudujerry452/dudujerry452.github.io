#!/usr/bin/env python3 

import matplotlib.pyplot as plt 
import random 
import numpy as np
import copy 
import plotly.tools as tls

class Ellipse: 
    def __init__(self, a, b, c1, c2, t): 
        self.args = np.array([a,b,c1,c2,t])

        self.rot = np.array([[np.cos(t), -np.sin(t)], [np.sin(t), np.cos(t)]])
        self.inv_rot = self.rot.T

    def update_rot(self): 
        a,b,c1,c2,t = self.args
        self.rot = np.array([[np.cos(t), -np.sin(t)], [np.sin(t), np.cos(t)]])
        self.inv_rot = self.rot.T

    def trs(self, x, y): 
        a,b,c1,c2,t = self.args
        pt = np.array([x, y]) 
        npt = self.rot @ pt + np.array([[c1], [c2]])
        return (npt[0], npt[1])
        
    def inv(self, x, y): 
        a,b,c1,c2,t = self.args
        pt = np.array([x, y])
        npt = self.inv_rot @ (pt - np.array([[c1], [c2]]))
        return (npt[0], npt[1])

    def chk(self, x, y): 
        a,b,c1,c2,t = self.args
        a = abs(a) + 1e-6 
        b = abs(b) + 1e-6
        nx, ny = self.inv(x, y) 
        nx = nx * nx / (a * a) 
        ny = ny * ny / (b * b)
        return not np.any(nx + ny - 1 > 0)

    def loss(self, x, y): 
        a,b,c1,c2,t = self.args
        a = abs(a) + 1e-6 
        b = abs(b) + 1e-6
        nx, ny = self.inv(x, y) 
        nx = nx * nx / (a * a) 
        ny = ny * ny / (b * b)
        dist = nx + ny - 1
        vio = np.maximum(0, dist)
        return np.sum(vio**2) * 30 + np.pi * a * b; 

    def draw_st(self, ax): 
        a,b,c1,c2,t = self.args
        angles = np.linspace(0, 2*np.pi, 100) 
        x = a*np.cos(angles) 
        y = b*np.sin(angles) 
        ax.plot(x, y)

    def draw_trs(self, ax): 
        a,b,c1,c2,t = self.args
        angles = np.linspace(0, 2*np.pi, 100) 
        x = a*np.cos(angles) 
        y = b*np.sin(angles) 
        ax.plot(*(self.trs(x, y))) 


def inputs(): 
    xs = []
    ys = []
    for i in range(20): 
        xs.append(random.random() * 1)
        ys.append(random.random() * 1)
    return (xs, ys) 

def gradient(xs, ys, elp): 
    eps = 1e-5 
    gd = []
    for i in range(len(elp.args)): 
        elp.args[i] -= eps 
        elp.update_rot()
        b = elp.loss(xs, ys)

        elp.args[i] += 2*eps 
        elp.update_rot()
        e = elp.loss(xs, ys)

        gd.append((e-b) / (2*eps)) 

        elp.args[i] -= eps 
        elp.args[0] = np.maximum(1e-6, elp.args[0])
        elp.args[1] = np.maximum(1e-6, elp.args[1])
        elp.update_rot()
    return gd 


def solve(xs, ys):

    losses = []
    min_loss = 999.0 
    min_elp = Ellipse(0,0,0,0,0)

    xa, xi, ya, yi = np.max(xs), np.min(xs), np.max(ys), np.min(ys)
    ctr = ((xa+xi)/2, (ya+yi)/2) 
    rad = np.sqrt((xa-xi)**2 + (ya-yi)**2)/2 

    elp = Ellipse(a=rad, b=rad, c1=ctr[0], c2=ctr[1], t=0) # init 
 
    print("init loss = ", elp.loss(xs, ys))

    max_epoch = 1000
    lr = 0.01
    for epoch in range(max_epoch): 
        gd = gradient(xs, ys, elp)
        elp.args += -lr * np.array(gd)
        elp.update_rot()
        tloss = elp.loss(xs, ys)
        if tloss < min_loss and elp.chk(xs, ys): 
            min_loss = tloss 
            min_elp = copy.deepcopy(elp)
        losses.append(elp.loss(xs, ys))

    return elp, elp.loss(xs, ys), min_elp, min_loss, losses;
